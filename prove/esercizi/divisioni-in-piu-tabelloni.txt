# DIVISIONI DEI GIOCATORI IN PIU' TABELLONI COLLEGATI, DAGLI ESERCIZI DEL MANUALE
# ===============================================================================
#
# Qui e' ricopiata, esercizio per esercizio, la divisione dei giocatori in piu'
# tabelloni scelta dal manuale (Volume I, capitolo VI; Volume II, capitolo 6).
# Le prove automatiche controllano che la divisione rispetti le regole, e che il
# programma proponga gli stessi qualificati uscenti.
#
# Ogni esercizio comincia con il suo nome tra parentesi quadre.
#   - "giocatori" e "qualificati entranti" (quelli che arrivano al primo
#     tabellone) sono i dati del problema;
#   - le righe "tabellone N" sono la soluzione del manuale, scritte come in
#     dati/torneo.txt: le classifiche di quel tabellone e i qualificati uscenti.
#     L'ultimo tabellone e' quello che segue (il tabellone finale, o quello di
#     terza categoria che l'esercizio da' come dato).
#   - "proposta del programma" sono i qualificati uscenti che il programma
#     propone, tabellone per tabellone, quando sono diversi da quelli del manuale
#     (senza righe scritte, un tabellone dopo l'altro).
#   - "proposta per categoria" e' la divisione che il programma propone quando
#     in dati/torneo.txt non ci sono righe "tabellone N" (un tabellone per
#     categoria), quando e' diversa da quella del manuale.
#
# Fonte: Volume I, capitolo VI, ipotesi 2 (pagine 84-93);
#        Volume II, esercizi da 6.01 a 6.06 (pagine 192-207).

[Volume I, capitolo VI, ipotesi 2]

tabellone 1 = 4.NC ; qualificati uscenti = 8

tabellone 4 = 2.8, 2.7, 2.6, 2.5, 2.4, 2.2, 2.1

[Volume II, esercizio 6.01]
giocatori = 4 (4.6), 3 (4.5), 4 (4.4), 4 (4.3), 2 (4.2), 2 (4.1), 3 (3.5), 1 (3.4)
qualificati entranti = 4
tabellone 1 = 4.6, 4.5 ; qualificati uscenti = 4
tabellone 2 = 4.4, 4.3 ; qualificati uscenti = 4
tabellone 3 = 4.2, 4.1 ; qualificati uscenti = 4

proposta per categoria = 4.6, 4.5, 4.4, 4.3, 4.2, 4.1 | 3.5, 3.4

# (il manuale scrive per errore "(3.3)" e "(3.2)").
[Volume II, esercizio 6.02]

tabellone 1 = 4.6, 4.5, 4.4 ; qualificati uscenti = 8
tabellone 2 = 4.3, 4.2 ; qualificati uscenti = 8
tabellone 3 = 4.1 ; qualificati uscenti = 8
tabellone 4 = 3.5, 3.4
proposta per categoria = 4.6, 4.5, 4.4, 4.3, 4.2, 4.1 | 3.5, 3.4
